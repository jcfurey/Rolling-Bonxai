cmake_minimum_required(VERSION 3.16)
project(rolling_bonxai_cloud)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rolling_bonxai_msgs REQUIRED)
find_library(ZSTD_LIBRARY zstd REQUIRED)
add_library(window_core SHARED src/cloud_window.cpp)
target_compile_features(window_core PUBLIC cxx_std_17)
target_compile_options(window_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(window_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(window_core PRIVATE ${ZSTD_LIBRARY})
add_library(window_ros SHARED src/window_conversion.cpp)
target_compile_features(window_ros PUBLIC cxx_std_17)
target_include_directories(window_ros PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(window_ros PUBLIC window_core ${sensor_msgs_TARGETS} ${rolling_bonxai_msgs_TARGETS})
add_executable(surface_window_receiver src/window_receiver.cpp)
target_link_libraries(surface_window_receiver PRIVATE window_ros rclcpp::rclcpp)
install(TARGETS window_core window_ros EXPORT windowTargets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS surface_window_receiver DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
ament_export_targets(windowTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(sensor_msgs rolling_bonxai_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_cloud_window test/test_cloud_window.cpp)
  target_link_libraries(test_cloud_window window_ros ${ZSTD_LIBRARY})
endif()
ament_package()
