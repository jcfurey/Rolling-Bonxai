cmake_minimum_required(VERSION 3.16)
project(rolling_bonxai_rviz)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rolling_bonxai_msgs REQUIRED)
# Match RViz's exported Qt major version, including installations with both.
get_target_property(_rviz_links rviz_common::rviz_common INTERFACE_LINK_LIBRARIES)
if("${_rviz_links}" MATCHES "Qt6::")
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  set(_window_qt Qt6::Widgets)
else()
  find_package(Qt5 REQUIRED COMPONENTS Widgets)
  set(_window_qt Qt5::Widgets)
endif()
set(CMAKE_AUTOMOC ON)
add_library(map_window_panel SHARED src/map_window_panel.cpp include/rolling_bonxai_rviz/map_window_panel.hpp)
target_compile_features(map_window_panel PUBLIC cxx_std_17)
target_include_directories(map_window_panel PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(map_window_panel PUBLIC rviz_common::rviz_common ${_window_qt} rclcpp::rclcpp
  ${std_msgs_TARGETS} ${rolling_bonxai_msgs_TARGETS} pluginlib::pluginlib)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS map_window_panel LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_ros REQUIRED)
  ament_add_ros_isolated_gtest(test_map_window_panel test/test_map_window_panel.cpp TIMEOUT 60)
  target_link_libraries(test_map_window_panel map_window_panel)
endif()
ament_package()
